     WITH TotalMinutesCTE AS (
        SELECT 
            -- Convert scheduled_hours string to total minutes
            CAST(LEFT(scheduled_hours, CHARINDEX(':', scheduled_hours) - 1) AS INT) * 60 +
            CAST(SUBSTRING(scheduled_hours, CHARINDEX(':', scheduled_hours) + 1, 2) AS INT) AS sched_mins,
        
            -- Convert booked_hours string to total minutes
            CAST(LEFT(booked_hours, CHARINDEX(':', booked_hours) - 1) AS INT) * 60 +
            CAST(SUBSTRING(booked_hours, CHARINDEX(':', booked_hours) + 1, 2) AS INT) AS booked_mins
        FROM [dbo].[BRONZE_ZENOTI_EMPLOYEE_SCHEDULES]
         where date >= '2026-06-01' and
	           date < '2026-07-05'    
    )
    SELECT 
        -- Formats to HHHH:MM
        CAST(SUM(sched_mins) / 60 AS VARCHAR(20)) + ':' + 
        RIGHT('0' + CAST(SUM(sched_mins) % 60 AS VARCHAR(2)), 2) AS total_scheduled_hours,

        -- Formats to HHHH:MM
        CAST(SUM(booked_mins) / 60 AS VARCHAR(20)) + ':' + 
        RIGHT('0' + CAST(SUM(booked_mins) % 60 AS VARCHAR(2)), 2) AS total_booked_hours
    FROM TotalMinutesCTE;

  DECLARE @s date = '2026-06-01', @e date = '2026-06-30';

;WITH esth_staff AS (          -- role defined by schedule.job_name
    SELECT DISTINCT employee_name
    FROM dbo.V_ZENOTI_EMPLOYEE_SCHEDULES
    WHERE CAST([date] AS DATE) BETWEEN @s AND @e
      AND LOWER(job_name) = 'esthetician'
      AND employee_name IS NOT NULL
)
SELECT
    rev                                   AS esthetician_revenue,
    hrs                                   AS esthetician_booked_hours,
    CASE WHEN hrs > 0 THEN rev / hrs END  AS rev_per_hour
FROM (
    SELECT
        -- revenue: sales_accrual.sold_by ∈ esthetician staff
        (SELECT SUM(sa.sales_exc_tax)
           FROM dbo.V_ZENOTI_SALES_ACCRUAL sa
           WHERE CAST(sa.sale_date AS DATE) BETWEEN @s AND @e
             AND sa.sold_by IN (SELECT employee_name FROM esth_staff)) AS rev,
        -- booked hours: schedule rows for those same esthetician staff
        (SELECT SUM(
                 CASE WHEN booked_hours LIKE '%:%'
                      THEN CAST(SUBSTRING(booked_hours,1,CHARINDEX(':',booked_hours)-1) AS FLOAT)
                         + CAST(SUBSTRING(booked_hours,CHARINDEX(':',booked_hours)+1,LEN(booked_hours)) AS FLOAT)/60.0
                      ELSE COALESCE(TRY_CAST(booked_hours AS FLOAT),0) END)
           FROM dbo.V_ZENOTI_EMPLOYEE_SCHEDULES
           WHERE CAST([date] AS DATE) BETWEEN @s AND @e
             AND employee_name IN (SELECT employee_name FROM esth_staff)) AS hrs ) x;

   SELECT 
    -- Sales Accrual Columns
    sa.sold_by,
    sa.sales_exc_tax,
    sa.sales_inc_tax,
    sa.sale_date AS sales_date, 

    -- Employee Schedule Columns
    es.employee_name,
    es.job_name AS schedule_job_name,
    es.scheduled_hours,
    es.booked_hours
FROM [dbo].[BRONZE_ZENOTI_SALES_ACCRUAL] sa
LEFT JOIN [dbo].[BRONZE_ZENOTI_EMPLOYEE_SCHEDULES] es 
    ON sa.sold_by = es.employee_name  
    AND sa.sale_date = es.date 
    AND es.job_name = 'Treatment Provider' -- Filters the schedules side before joining
WHERE 
    sa.sale_date >= '2026-06-01' 
    AND sa.sale_date < '2026-07-01'
ORDER BY 
    sa.sale_date DESC;
